set(GIGE_MEL_SANITIZERS "" CACHE STRING "Match host sanitizer configuration")
option(GIGE_MEL_MODULE_NEGATIVE_TESTS "Build deliberately invalid test modules" OFF)
if(NOT EXISTS "${GIGE_MEL_SOURCE_ROOT}/src/profiles/profile_module.hpp")
  message(FATAL_ERROR "Set GIGE_MEL_SOURCE_ROOT to the matching provider checkout")

    "${GIGE_MEL_SOURCE_ROOT}/src/profiles/camera_profile.cpp"

set_target_properties(representation PROPERTIES POSITION_INDEPENDENT_CODE ON)
function(module_options target)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE "${GIGE_MEL_SOURCE_ROOT}/include"
    "${GIGE_MEL_SOURCE_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fvisibility=hidden)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(${target} PRIVATE -fno-gnu-unique)
  endif()
  if(GIGE_MEL_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=${GIGE_MEL_SANITIZERS} -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=${GIGE_MEL_SANITIZERS})
  endif()
endfunction()
module_options(representation)
function(profile_module target variant)
  add_library(${target} MODULE module.cpp)
  module_options(${target})
  target_compile_definitions(${target} PRIVATE MODULE_VARIANT=${variant})
  target_link_libraries(${target} PRIVATE representation)
  target_link_options(${target} PRIVATE "LINKER:-z,defs" "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endfunction()
profile_module(rate_limited_profile 0)
# GIGE-MEL-047: test-only profile whose state implements the optional navigation
# handler. Same v1 descriptor, symbol, flags and identity rules as above.
function(navigation_module target variant)
  add_library(${target} MODULE navigation_module.cpp)
  module_options(${target})
  target_compile_definitions(${target} PRIVATE NAVIGATION_MODULE_VARIANT=${variant})
  target_link_libraries(${target} PRIVATE representation)
  target_link_options(${target} PRIVATE "LINKER:-z,defs" "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endfunction()
