cmake_minimum_required(VERSION 3.24)
project(promise_thread_exit_repro LANGUAGES CXX)
find_package(Threads REQUIRED)
set(REPRO_SANITIZERS "thread" CACHE STRING "Sanitizers; empty for unsanitized control")
add_executable(promise_thread_exit promise_thread_exit.cpp)
target_compile_features(promise_thread_exit PRIVATE cxx_std_20)
target_compile_options(promise_thread_exit PRIVATE -g -O0 -Wall -Wextra -Wpedantic -Werror -fno-omit-frame-pointer)
target_link_libraries(promise_thread_exit PRIVATE Threads::Threads)
if(REPRO_SANITIZERS)
  target_compile_options(promise_thread_exit PRIVATE "-fsanitize=${REPRO_SANITIZERS}")
  target_link_options(promise_thread_exit PRIVATE "-fsanitize=${REPRO_SANITIZERS}")
endif()
option(REPRO_PROVIDER "Also build the SDK-free installed-provider ownership reproduction" OFF)
if(REPRO_PROVIDER)
  find_package(GigeMelProvider CONFIG REQUIRED)
  add_executable(provider_owners provider_owners.cpp)
  target_link_libraries(provider_owners PRIVATE gige_mel::provider Threads::Threads)
  target_compile_options(provider_owners PRIVATE -g -O0 -Wall -Wextra -Wpedantic -Werror -fno-omit-frame-pointer)
  if(REPRO_SANITIZERS)
    target_compile_options(provider_owners PRIVATE "-fsanitize=${REPRO_SANITIZERS}")
    target_link_options(provider_owners PRIVATE "-fsanitize=${REPRO_SANITIZERS}")
  endif()
endif()
